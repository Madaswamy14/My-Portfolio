Git Automation Conventional Commits Changelog Gen Python
def parse_commits(commit_lines): """Parses lines into categories based on Conventional Commits.""" categorized = defaultdict(list) for line in commit_lines: match = COMMIT_PATTERN.match(line) if match: c_type, c_scope, c_desc = match.groups() # Groups by type (e.g. 'feat' -> '✨ New Features') header = TYPE_MAP.get(c_type, '🛠 Other Changes') categorized[header].append(c_desc) return categorized

def process_screenshots(config_file, output_dir, headless=True): """Reads a JSON config and captures screenshots via Selenium.""" driver = setup_driver(headless) with open(config_file) as f: shots = json.load(f) for item in shots: driver.set_window_size(item['width'], item['height']) driver.get(item['url']) driver.save_screenshot(f"{output_dir}/{item['output']}") driver.quit()
